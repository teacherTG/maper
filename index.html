                        pt.y = (pt.y / oldH) * currentCourtH;

                    }

                };

                polygons.forEach(poly => poly.forEach(updateScale));

                currentPolygon.forEach(updateScale);

                

                items.forEach(item => {

                    item.x = (item.x / oldW) * currentCourtW;

                    item.y = (item.y / oldH) * currentCourtH;

                });

                drawForeground();

            }

        });


        function drawLine(ctx, x1, y1, x2, y2, color = 'white', width = 2) {

            ctx.beginPath();

            ctx.moveTo(x1, y1);

            ctx.lineTo(x2, y2);

            ctx.strokeStyle = color;

            ctx.lineWidth = width;

            ctx.stroke();

        }


        function drawCircle(ctx, x, y, r, color = 'white', width = 2, fill = false) {

            // 원 반지름이 0 이하가 되지 않도록 철저히 방어 (IndexSizeError 방지)

            r = Math.max(0.1, Math.abs(r)); 

            

            ctx.beginPath();

            ctx.arc(x, y, r, 0, Math.PI * 2);

            if (fill) {

                ctx.fillStyle = color;

                ctx.fill();

            } else {

                ctx.strokeStyle = color;

                ctx.lineWidth = width;

                ctx.stroke();

            }

        }

                    const margin = 20 * currentScale;

                    const homeX = w - margin;

                    const homeY = h - margin;

                    

                    // 야구장 스케일 기준 크기 (루간 거리 약 27.43m 비율 기반)

                    const baseDist = 27 * cs; 

                    

                    // 내야 흙(Dirt) 그리기 (다이아몬드 + 투수 마운드 주변 크게)

                    const dirtRadius = baseDist + (5 * cs);

                    bgCtx.beginPath();

                    bgCtx.moveTo(homeX, homeY);

                    bgCtx.arc(homeX, homeY, dirtRadius, Math.PI, 1.5 * Math.PI);

                    bgCtx.lineTo(homeX, homeY);

                    bgCtx.fillStyle = '#b4835a'; // 흙 색상

                    bgCtx.fill();

                    

                    // 1, 2, 3루 위치 계산 (1루는 위쪽, 3루는 왼쪽)

                    const firstX = homeX;

                    const firstY = homeY - baseDist;

                    const thirdX = homeX - baseDist;

                    const thirdY = homeY;

                    const secondX = homeX - baseDist;

                    const secondY = homeY - baseDist;

                    

                    // 내야 다이아몬드 잔디

                    bgCtx.beginPath();

                    bgCtx.moveTo(homeX, homeY);

                    bgCtx.lineTo(firstX, firstY);

                    bgCtx.lineTo(secondX, secondY);

                    bgCtx.lineTo(thirdX, thirdY);

                    bgCtx.closePath();

                    bgCtx.fillStyle = '#2b7a2b'; // 짙은 잔디색

                    bgCtx.fill();

                    bgCtx.strokeStyle = '#b4835a'; // 베이스 라인 흙길 표현

                    bgCtx.lineWidth = 4 * currentScale;

                    bgCtx.stroke();

                    

                    // 파울 라인 (1루, 3루 선상 연장선)

                    const foulLineLen = Math.max(w, h);

                    drawLine(bgCtx, homeX, homeY, homeX, homeY - foulLineLen, 'white', 2); // 1루측 우익선상

                    drawLine(bgCtx, homeX, homeY, homeX - foulLineLen, homeY, 'white', 2); // 3루측 좌익선상

                    

                    // 투수 마운드

                    const moundX = homeX - (baseDist / 2);

                    const moundY = homeY - (baseDist / 2);

                    drawCircle(bgCtx, moundX, moundY, 3 * cs, '#b4835a', 1, true); // 마운드 흙

                    bgCtx.fillStyle = 'white';

                    bgCtx.fillRect(moundX - 1*cs, moundY - 0.3*cs, 2*cs, 0.6*cs); // 투수판

                    

                    // 베이스 그리기 (흰색 사각형)

                    const baseSize = 1.5 * cs;

                    bgCtx.fillStyle = 'white';

                    bgCtx.fillRect(firstX - baseSize/2, firstY - baseSize/2, baseSize, baseSize); // 1루

                    bgCtx.fillRect(secondX - baseSize/2, secondY - baseSize/2, baseSize, baseSize); // 2루

                    bgCtx.fillRect(thirdX - baseSize/2, thirdY - baseSize/2, baseSize, baseSize); // 3루

                    

                    // 홈플레이트 (오각형)

                    bgCtx.beginPath();

                    const hSize = 1 * cs;

                    bgCtx.moveTo(homeX, homeY);

                    bgCtx.lineTo(homeX, homeY - hSize);

                    bgCtx.lineTo(homeX - hSize, homeY - hSize);

                    bgCtx.lineTo(homeX - hSize, homeY);

                    bgCtx.closePath();

                    bgCtx.fillStyle = 'white';

                    bgCtx.fill();

                }

            } catch (err) {

                console.error("Court drawing failed:", err);

            }

            

            bgCtx.restore(); // 회전된 렌더링 컨텍스트 복구

        }


        function getResolvedPolyPoints(poly) {

            // 특정 다각형 배열의 점 실시간 좌표 헬퍼 함수

            return poly.map((pt, index) => {

                if (pt.ref) {

                    const item = items.find(i => i.id === pt.ref);

                    return item ? { x: item.x, y: item.y, ref: pt.ref, originalIndex: index } : null;

                }

                return { x: pt.x, y: pt.y, originalIndex: index };

            }).filter(pt => pt !== null);

        }


        function drawForeground() {

            const physW = isVerticalView ? currentCourtH : currentCourtW;

            const physH = isVerticalView ? currentCourtW : currentCourtH;

            fgCtx.clearRect(0, 0, physW, physH);


            // 🌟 세로 모드일 경우 캔버스 렌더링 컨텍스트를 90도 회전

            fgCtx.save();

            if (isVerticalView) {

                fgCtx.translate(currentCourtH, 0);

                fgCtx.rotate(Math.PI / 2);

            }


            // 헬퍼 함수: 다각형 그리기 (완성된 것과 현재 진행 중인 것 모두 처리)

            const drawPoly = (poly, isClosed, polyIndex = -1) => {

                const resolved = getResolvedPolyPoints(poly);

                if (resolved.length === 0) return;


                // 🌟 팀별 자동 색상 지정 로직

                let attackCount = 0;

                let oppCount = 0;

                for (const pt of poly) {

                    if (pt.ref) {

                        const item = items.find(i => i.id === pt.ref);

                        if (item && item.type === 'attacker') attackCount++;

                        if (item && item.type === 'opponent') oppCount++;

                    }

                }


                let strokeColor = config.lineColor; // 기본 파란선

                let fillColor = config.fillColor; // 기본 파란면

                

                if (attackCount > 0 && attackCount >= oppCount) {

                    // 공격수가 주를 이루면 붉은색 계열로 자동 변경

                    strokeColor = '#ef4444'; 

                    fillColor = 'rgba(239, 68, 68, 0.25)';

                } else if (oppCount > 0 && oppCount > attackCount) {

                    // 수비수가 주를 이루면 짙은 푸른색 계열로 자동 변경

                    strokeColor = '#2563eb'; 

                    fillColor = 'rgba(37, 99, 235, 0.25)';

                }


                fgCtx.beginPath();

                fgCtx.moveTo(resolved[0].x, resolved[0].y);

                for (let i = 1; i < resolved.length; i++) {

                    fgCtx.lineTo(resolved[i].x, resolved[i].y);

                }


                if (isClosed) {

                    fgCtx.closePath();

                    fgCtx.fillStyle = fillColor;

                    fgCtx.fill();

                }


                fgCtx.strokeStyle = strokeColor;

                fgCtx.lineWidth = config.lineWidth;

                fgCtx.lineJoin = 'round';

                fgCtx.stroke();


                // Draw Points

                for (let i = 0; i < resolved.length; i++) {

                    const p = resolved[i];

                    if (p.ref) continue; 


                    fgCtx.beginPath();

                    fgCtx.arc(p.x, p.y, config.pointRadius, 0, Math.PI * 2);

                    

                    let isHovered = false;

                    let isDragged = false;

                    

                    if (polyIndex === -1) {

                        // 현재 그리고 있는 다각형의 점

                        isHovered = hoveredElement && hoveredElement.type === 'currentPoint' && hoveredElement.index === p.originalIndex;

                        isDragged = draggingElement && draggingElement.type === 'currentPoint' && draggingElement.index === p.originalIndex;

                    } else {

                        // 이미 완성된 다각형의 점

                        isHovered = hoveredElement && hoveredElement.type === 'polyPoint' && hoveredElement.polyIndex === polyIndex && hoveredElement.ptIndex === p.originalIndex;

                        isDragged = draggingElement && draggingElement.type === 'polyPoint' && draggingElement.polyIndex === polyIndex && draggingElement.ptIndex === p.originalIndex;

                    }


                    if (isHovered || isDragged) {

                        fgCtx.fillStyle = config.hoverColor;

                        fgCtx.arc(p.x, p.y, config.pointRadius + 2, 0, Math.PI * 2); 

                    } else if (!isClosed && i === 0 && resolved.length > 2) {

                        fgCtx.fillStyle = '#10b981'; // Green to suggest closing

                    } else {

                        fgCtx.fillStyle = config.pointColor;

                    }

                    

                    fgCtx.fill();

                    fgCtx.strokeStyle = 'white';

                    fgCtx.lineWidth = 1.5;

                    fgCtx.stroke();

                }

            };


            // 1. 완성된 다각형들 모두 그리기

            polygons.forEach((poly, index) => drawPoly(poly, true, index));

            // 2. 현재 그리고 있는 다각형 그리기

            drawPoly(currentPolygon, false, -1);


            // 3. Draw Items (Ball, Attacker, Opponent)

            fgCtx.textAlign = 'center';

            fgCtx.textBaseline = 'middle';

            

            for (let i = 0; i < items.length; i++) {

                const item = items[i];

                const isHovered = hoveredElement && hoveredElement.type === 'item' && hoveredElement.index === i;

                const isDragged = draggingElement && draggingElement.type === 'item' && draggingElement.index === i;

                

                // 현재 그리고 있는 다각형의 시작점인지 확인

                const isStartPoint = currentPolygon.length > 2 && currentPolygon[0].ref === item.id;


                if (isStartPoint) {

                    fgCtx.beginPath();

                    fgCtx.arc(item.x, item.y, 22, 0, Math.PI * 2);

                    fgCtx.fillStyle = 'rgba(16, 185, 129, 0.4)'; 

                    fgCtx.fill();

                } else if (isHovered && currentMode === 'polygon') {

                    fgCtx.beginPath();

                    fgCtx.arc(item.x, item.y, 22, 0, Math.PI * 2);

                    fgCtx.fillStyle = 'rgba(59, 130, 246, 0.4)'; 

                    fgCtx.fill();

                } else if (isHovered || isDragged) {

                    fgCtx.beginPath();

                    fgCtx.arc(item.x, item.y, 22, 0, Math.PI * 2); 

                    fgCtx.fillStyle = 'rgba(0,0,0,0.15)'; 

                    fgCtx.fill();

                }


                if (item.type === 'ball') {

                    const t = courtSelect.value;

                    if (t === 'ultimate') {

                        // 얼티밋 프리즈비 디스크

                        fgCtx.beginPath();

                        fgCtx.arc(item.x, item.y, 18, 0, Math.PI * 2);

                        fgCtx.fillStyle = '#fef08a'; 

                        fgCtx.fill();

                        fgCtx.strokeStyle = '#ca8a04';

                        fgCtx.lineWidth = 3.5;

                        fgCtx.stroke();

                        

                        fgCtx.beginPath();

                        fgCtx.arc(item.x, item.y, 9, 0, Math.PI * 2);

                        fgCtx.strokeStyle = '#ca8a04';

                        fgCtx.lineWidth = 2.5;

                        fgCtx.stroke();

                    } else {

                        let emoji = '⚽';

                        if (t === 'basketball') emoji = '🏀';

                        if (t === 'volleyball' || t === 'netball') emoji = '🏐';

                        if (t === 'baseball') emoji = '⚾';

                        

                        // 공 이모지는 화면 회전과 무관하게 똑바로 보이도록 컨텍스트 역회전 처리

                        fgCtx.save();

                        fgCtx.translate(item.x, item.y);

                        if (isVerticalView) fgCtx.rotate(-Math.PI / 2);

                        fgCtx.globalAlpha = 1.0;

                        fgCtx.fillStyle = '#000000';

                        fgCtx.font = '39px Arial'; 

                        fgCtx.fillText(emoji, 0, 0);

                        fgCtx.restore();

                    }

                } else if (item.type === 'attacker') {

                    // 공격수: ⭕ (빨간색)

                    fgCtx.beginPath();

                    fgCtx.arc(item.x, item.y, 13, 0, Math.PI * 2);

                    fgCtx.fillStyle = '#ef4444'; 

                    fgCtx.fill();

                    fgCtx.strokeStyle = 'white';

                    fgCtx.lineWidth = 2.5;

                    fgCtx.stroke();

                    

                    fgCtx.beginPath();

                    fgCtx.arc(item.x, item.y, 5.5, 0, Math.PI * 2);

                    fgCtx.strokeStyle = 'white';

                    fgCtx.lineWidth = 3;

                    fgCtx.stroke();

                } else if (item.type === 'opponent') {

                    // 수비수: ❌ (파란색)

                    fgCtx.beginPath();

                    fgCtx.arc(item.x, item.y, 13, 0, Math.PI * 2); 

                    fgCtx.fillStyle = '#2563eb'; 

                    fgCtx.fill();

                    fgCtx.strokeStyle = 'white';

                    fgCtx.lineWidth = 2.5;

                    fgCtx.stroke();

                    

                    fgCtx.beginPath();

                    const d = 5.5; 

                    fgCtx.moveTo(item.x - d, item.y - d);

                    fgCtx.lineTo(item.x + d, item.y + d);

                    fgCtx.moveTo(item.x + d, item.y - d);

                    fgCtx.lineTo(item.x - d, item.y + d);

                    fgCtx.strokeStyle = 'white';

                    fgCtx.lineWidth = 3;

                    fgCtx.stroke();

                }

            }


            // 4. 선(Edge) 위에 마우스를 올렸을 때 점 추가 미리보기 (초록색 점) 표시

            if (hoveredElement && hoveredElement.type === 'polyEdge') {

                fgCtx.beginPath();

                fgCtx.arc(hoveredElement.insertX, hoveredElement.insertY, config.pointRadius, 0, Math.PI * 2);

                fgCtx.fillStyle = '#10b981'; // Green

                fgCtx.fill();

                fgCtx.strokeStyle = 'white';

                fgCtx.lineWidth = 1.5;

                fgCtx.stroke();

            }

            

            fgCtx.restore(); // 회전된 렌더링 컨텍스트 복구

            

            // --- UI 툴바 버튼 동적 업데이트 (면 완성/취소 버튼 표시) ---

            if (currentMode === 'polygon' && currentPolygon.length > 0) {

                finishPolygonBtn.classList.remove('hidden');

                if (currentPolygon.length >= 3) {

                    finishPolygonBtn.innerHTML = '✅ 면 완성';

                    finishPolygonBtn.className = 'px-3 py-2 text-sm font-bold text-white rounded-md transition-colors whitespace-nowrap snap-start active:scale-95 shadow-sm mx-1 bg-green-500 hover:bg-green-600';

                } else {

                    finishPolygonBtn.innerHTML = '❌ 선 끊기';

                    finishPolygonBtn.className = 'px-3 py-2 text-sm font-bold text-white rounded-md transition-colors whitespace-nowrap snap-start active:scale-95 shadow-sm mx-1 bg-red-500 hover:bg-red-600';

                }

            } else {

                finishPolygonBtn.classList.add('hidden');

                finishPolygonBtn.className = 'hidden'; 

            }

        }


        function getMousePos(e) {

            const rect = fgCanvas.getBoundingClientRect();

            let clientX = e.clientX;

            let clientY = e.clientY;

            

            if (e.touches && e.touches.length > 0) {

                clientX = e.touches[0].clientX;

                clientY = e.touches[0].clientY;

            }


            // 터치/클릭 지점의 화면상 좌표

            let sx = clientX - rect.left;

            let sy = clientY - rect.top;


            // 세로 모드로 회전되었을 때는 좌표를 논리적(가로 모드) 좌표계로 역변환하여 반환

            if (isVerticalView) {

                return {

                    x: sy,

                    y: currentCourtH - sx

                };

            } else {

                return {

                    x: sx,

                    y: sy

                };

            }

        }


        // 두 점(v, w)을 잇는 선분과 특정 점(p) 사이의 최단 거리 계산

        function distToSegmentSquared(p, v, w) {

            const l2 = (v.x - w.x) ** 2 + (v.y - w.y) ** 2;

            if (l2 === 0) return (p.x - v.x) ** 2 + (p.y - v.y) ** 2;

            let t = ((p.x - v.x) * (w.x - v.x) + (p.y - v.y) * (w.y - v.y)) / l2;

            t = Math.max(0, Math.min(1, t));

            return (p.x - (v.x + t * (w.x - v.x))) ** 2 + (p.y - (v.y + t * (w.y - v.y))) ** 2;

        }


        function distToSegment(p, v, w) {

            return Math.sqrt(distToSegmentSquared(p, v, w));

        }


        function checkIntersection(x, y) {

            // 1. Check items first (top layer)

            const itemRadius = 26; 

            for (let i = items.length - 1; i >= 0; i--) {

                const item = items[i];

                if (Math.hypot(item.x - x, item.y - y) < itemRadius) {

                    return { type: 'item', index: i };

                }

            }

            

            const pointHitZone = config.pointRadius + 10; 

            

            // 2. Check current ongoing polygon points

            for (let i = currentPolygon.length - 1; i >= 0; i--) {

                const p = currentPolygon[i];

                if (p.ref) continue; 

                if (Math.hypot(p.x - x, p.y - y) < pointHitZone) {

                    return { type: 'currentPoint', index: i };

                }

            }


            // 3. Check closed polygons points

            for (let pIdx = polygons.length - 1; pIdx >= 0; pIdx--) {

                const poly = polygons[pIdx];

                for (let i = poly.length - 1; i >= 0; i--) {

                    const p = poly[i];

                    if (p.ref) continue;

                    if (Math.hypot(p.x - x, p.y - y) < pointHitZone) {

                        return { type: 'polyPoint', polyIndex: pIdx, ptIndex: i };

                    }

                }

            }

            

            // 4. 다각형의 선(Edge)을 클릭해서 새로운 점을 추가할 수 있도록 선분 교차 확인

            if (currentMode === 'polygon' || currentMode === 'move') {

                const edgeHitZone = 8; // 선 클릭 판정 범위

                const pObj = { x, y };


                // 완성된 다각형들의 선

                for (let pIdx = polygons.length - 1; pIdx >= 0; pIdx--) {

                    const poly = getResolvedPolyPoints(polygons[pIdx]);

                    for (let i = 0; i < poly.length; i++) {

                        const p1 = poly[i];

                        const p2 = poly[(i + 1) % poly.length];

                        const dist = distToSegment(pObj, p1, p2);

                        if (dist < edgeHitZone) {

                            // 시각적 피드백을 위해 선 위로 수선의 발(가장 가까운 점)을 투영

                            const l2 = (p1.x - p2.x) ** 2 + (p1.y - p2.y) ** 2;

                            let t = ((x - p1.x) * (p2.x - p1.x) + (y - p1.y) * (p2.y - p1.y)) / l2;

                            t = Math.max(0, Math.min(1, t));

                            return { 

                                type: 'polyEdge', 

                                polyIndex: pIdx, 

                                insertIndex: p1.originalIndex + 1, 

                                insertX: p1.x + t * (p2.x - p1.x), 

                                insertY: p1.y + t * (p2.y - p1.y) 

                            };

                        }

                    }

                }

                

                // 현재 그리고 있는 다각형의 선

                if (currentPolygon.length > 1) {

                    const curPoly = getResolvedPolyPoints(currentPolygon);

                    for (let i = 0; i < curPoly.length - 1; i++) {

                        const p1 = curPoly[i];

                        const p2 = curPoly[i + 1];

                        const dist = distToSegment(pObj, p1, p2);

                        if (dist < edgeHitZone) {

                            const l2 = (p1.x - p2.x) ** 2 + (p1.y - p2.y) ** 2;

                            let t = ((x - p1.x) * (p2.x - p1.x) + (y - p1.y) * (p2.y - p1.y)) / l2;

                            t = Math.max(0, Math.min(1, t));

                            return { 

                                type: 'polyEdge', 

                                polyIndex: -1, 

                                insertIndex: p1.originalIndex + 1, 

                                insertX: p1.x + t * (p2.x - p1.x), 

                                insertY: p1.y + t * (p2.y - p1.y) 

                            };

                        }

                    }

                }

            }


            return null;

        }

        function handlePointerDown(e) {

            if (e.type === 'touchstart') e.preventDefault();


            const pos = getMousePos(e);

            lastPos = pos;

            const intersected = checkIntersection(pos.x, pos.y);


            isDragged = false; 


            if (currentMode === 'eraser') {

                if (intersected) {

                    if (intersected.type === 'currentPoint') {

                        currentPolygon.splice(intersected.index, 1);

                    } else if (intersected.type === 'polyPoint') {

                        polygons[intersected.polyIndex].splice(intersected.ptIndex, 1);

                        if (polygons[intersected.polyIndex].length < 3) {

                            polygons.splice(intersected.polyIndex, 1); // 점이 3개 미만이 되면 다각형 자체를 삭제

                        }

                    } else if (intersected.type === 'item') {

                        const itemId = items[intersected.index].id;

                        items.splice(intersected.index, 1);

                        

                        // 지워진 아이템(선수)이 포함된 다각형 연결 정보도 정리

                        polygons = polygons.map(poly => poly.filter(pt => pt.ref !== itemId)).filter(poly => poly.length >= 3);

                        currentPolygon = currentPolygon.filter(pt => pt.ref !== itemId);

                    }

                    saveState();

                    hoveredElement = null;

                }

                drawForeground();

                return;

            }


            // '이동' 전용 모드 처리

            if (currentMode === 'move') {

                if (intersected) {

                    draggingElement = intersected;

                    fgCanvas.classList.add('grabbing');

                    fgCanvas.classList.remove('grab');

                }

                drawForeground();

                return;

            }


            // 선을 클릭해서 새로운 점을 추가하는 동작

            if (intersected && intersected.type === 'polyEdge') {

                const newPt = { id: Date.now() + Math.random(), x: intersected.insertX, y: intersected.insertY };

                if (intersected.polyIndex === -1) {

                    currentPolygon.splice(intersected.insertIndex, 0, newPt);

                    draggingElement = { type: 'currentPoint', index: intersected.insertIndex };

                } else {

                    polygons[intersected.polyIndex].splice(intersected.insertIndex, 0, newPt);

                    draggingElement = { type: 'polyPoint', polyIndex: intersected.polyIndex, ptIndex: intersected.insertIndex };

                }

                isDragged = true; // 강제로 드래그 상태로 만들어 기록되도록 함

                fgCanvas.classList.add('grabbing');

                fgCanvas.classList.remove('pointer', 'grab');

                drawForeground();

                return;

            }


            // 다각형, 공, 수비수, 공격수 추가 모드 처리

            if (currentMode === 'polygon') {

                // 다각형 닫기 (시작점 클릭 확인)

                let isStartPointClicked = false;

                if (currentPolygon.length > 2) {

                    if (intersected && intersected.type === 'item' && currentPolygon[0].ref === items[intersected.index].id) {

                        isStartPointClicked = true; // 시작점이 선수(아이템)인 경우

                    } else if (intersected && intersected.type === 'currentPoint' && intersected.index === 0) {

                        isStartPointClicked = true; // 시작점이 일반 점인 경우

                    }

                }


                if (isStartPointClicked) {

                    // 시작점을 눌렀으므로 면을 닫아 완성합니다

                    polygons.push([...currentPolygon]);

                    currentPolygon = [];

                    saveState();

                } else {

                    // 선수 위에 클릭하면 점을 선수에게 편입(Snap), 아니면 빈 공간에 새 점 생성

                    if (intersected && intersected.type === 'item') {

                        const snappedItemId = items[intersected.index].id;

                        currentPolygon.push({ id: Date.now() + Math.random(), ref: snappedItemId });

                    } else {

                        currentPolygon.push({ id: Date.now() + Math.random(), x: pos.x, y: pos.y });

                    }

                    // 추가한 점을 떼지 않고 즉시 드래그할 수 있도록 설정

                    draggingElement = { type: 'currentPoint', index: currentPolygon.length - 1 };

                    saveState();

                }

            } else if (currentMode === 'ball') {

                items.push({ id: Date.now() + Math.random(), type: 'ball', x: pos.x, y: pos.y });

                draggingElement = { type: 'item', index: items.length - 1 }; // 클릭 즉시 드래그 가능

                saveState();

            } else if (currentMode === 'opponent') {

                items.push({ id: Date.now() + Math.random(), type: 'opponent', x: pos.x, y: pos.y });

                draggingElement = { type: 'item', index: items.length - 1 }; // 클릭 즉시 드래그 가능

                saveState();

            } else if (currentMode === 'attacker') {

                items.push({ id: Date.now() + Math.random(), type: 'attacker', x: pos.x, y: pos.y });

                draggingElement = { type: 'item', index: items.length - 1 }; // 클릭 즉시 드래그 가능

                saveState();

            }

            

            drawForeground();

        }


        function handlePointerMove(e) {

            if (e.type === 'touchmove') e.preventDefault();

            const pos = getMousePos(e);

            lastPos = pos;


            if (draggingElement) {

                if (draggingElement.type === 'currentPoint') {

                    currentPolygon[draggingElement.index].x = pos.x;

                    currentPolygon[draggingElement.index].y = pos.y;

                } else if (draggingElement.type === 'polyPoint') {

                    polygons[draggingElement.polyIndex][draggingElement.ptIndex].x = pos.x;

                    polygons[draggingElement.polyIndex][draggingElement.ptIndex].y = pos.y;

                } else if (draggingElement.type === 'item') {

                    items[draggingElement.index].x = pos.x;

                    items[draggingElement.index].y = pos.y;

                }

                isDragged = true;

                drawForeground();

            } else {

                const intersected = checkIntersection(pos.x, pos.y);

                const sameElement = (a, b) => {

                    if (!a && !b) return true;

                    if (!a || !b) return false;

                    if (a.type !== b.type) return false;

                    if (a.type === 'polyPoint') return a.polyIndex === b.polyIndex && a.ptIndex === b.ptIndex;

                    return a.index === b.index;

                };


                if (!sameElement(intersected, hoveredElement)) {

                    hoveredElement = intersected;

                    fgCanvas.classList.remove('pointer', 'grab'); // 기존 커서 초기화

                    if (hoveredElement) {

                        if (currentMode === 'move' && hoveredElement.type !== 'polyEdge') {

                            fgCanvas.classList.add('grab'); // 이동 모드일 때는 손바닥

                        } else {

                            fgCanvas.classList.add('pointer'); // 다른 모드나 선 위일 때는 포인터

                        }

                    }

                    drawForeground();

                }

            }

        }


        function handlePointerUp() {

            if (draggingElement) {

                if (isDragged) {

                    // 점을 드래그하다가 다른 선수 위에서 놓았을 때 자동으로 선수에게 '편입(Snap)'되도록 처리

                    if (draggingElement.type === 'polyPoint' || draggingElement.type === 'currentPoint') {

                        const itemRadius = 26;

                        let snappedItemId = null;

                        for (let i = items.length - 1; i >= 0; i--) {

                            if (Math.hypot(items[i].x - lastPos.x, items[i].y - lastPos.y) < itemRadius) {

                                snappedItemId = items[i].id;

                                break;

                            }

                        }

                        if (snappedItemId) {

                            if (draggingElement.type === 'polyPoint') {

                                polygons[draggingElement.polyIndex][draggingElement.ptIndex] = { id: Date.now() + Math.random(), ref: snappedItemId };

                            } else if (draggingElement.type === 'currentPoint') {

                                currentPolygon[draggingElement.index] = { id: Date.now() + Math.random(), ref: snappedItemId };

                            }

                        }

                    }

                    saveState(); 

                }

                draggingElement = null;

                fgCanvas.classList.remove('grabbing');

                

                // 마우스를 놓았을 때 여전히 요소 위에 있다면 손바닥 모양 유지

                if (currentMode === 'move' && hoveredElement) {

                    fgCanvas.classList.add('grab');

                }

                

                drawForeground();

            }

        }
